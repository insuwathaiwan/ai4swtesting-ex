## Prompt

```text
Create tests for a file upload feature
```

## Response

```ts
/**
 * upload.test.ts
 *
 * Vitest suite for a file upload endpoint (Next.js App Router route handler).
 *
 * ASSUMPTIONS (adjust the constants/imports below to match your real code):
 *   - Route handler:  app/api/upload/route.ts  exporting  POST(request: Request)
 *   - Storage layer:  lib/storage.ts           exporting  saveFile(name, buffer, mimeType) -> { url }
 *   - Request:        multipart/form-data with the file in the field named "file"
 *   - Success:        201 + JSON { url, name, size }
 *   - Errors:         400 (bad input), 413 (too large), 415 (type not allowed), 500 (storage failure)
 *   - Limits:         max 5 MB; allowed types png / jpeg / pdf
 *
 * Install:  npm i -D vitest
 * Run:      npx vitest run upload.test.ts
 *
 * Test design (Jorgensen-style):
 *   - Equivalence classes: valid type / invalid type / empty / oversize
 *   - Boundary value analysis on size: MAX-1, MAX, MAX+1, 0, 1
 *   - Robustness / security: spoofed MIME, double extension, path traversal, null byte
 */

import { describe, it, expect, vi, beforeEach } from "vitest";

// ---------------------------------------------------------------------------
// Mock the storage layer so tests never touch disk / S3.
// vi.mock is hoisted, so it applies before the route is imported.
// ---------------------------------------------------------------------------
vi.mock("@/lib/storage", () => ({
  saveFile: vi.fn(),
}));

import { POST } from "@/app/api/upload/route";
import { saveFile } from "@/lib/storage";

// ---------------------------------------------------------------------------
// Constants: keep in sync with the implementation
// ---------------------------------------------------------------------------
const MAX_SIZE = 5 * 1024 * 1024; // 5 MB

// Magic-byte headers so "real" files pass content sniffing (if you implement it)
const PNG_HEADER = [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a];
const JPEG_HEADER = [0xff, 0xd8, 0xff];
const PDF_HEADER = [0x25, 0x50, 0x44, 0x46, 0x2d]; // "%PDF-"

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------

/** Build a byte array of an exact size, starting with a given header. */
function makeBytes(size: number, header: number[] = PNG_HEADER): Uint8Array {
  const bytes = new Uint8Array(size);
  bytes.set(header.slice(0, Math.min(header.length, size)));
  return bytes;
}

/** Build a File object (global in Node 20+). */
function makeFile(name: string, type: string, bytes: Uint8Array): File {
  return new File([bytes], name, { type });
}

/** Build a multipart Request. `fieldName` lets us test a missing/wrong field. */
function makeUploadRequest(files: File[], fieldName = "file"): Request {
  const form = new FormData();
  for (const f of files) form.append(fieldName, f);
  return new Request("http://localhost/api/upload", {
    method: "POST",
    body: form, // fetch sets the multipart boundary header automatically
  });
}

// Convenience valid fixtures
const validPng = (size = 1024) =>
  makeFile("photo.png", "image/png", makeBytes(size, PNG_HEADER));
const validPdf = (size = 1024) =>
  makeFile("doc.pdf", "application/pdf", makeBytes(size, PDF_HEADER));

// ---------------------------------------------------------------------------
// Tests
// ---------------------------------------------------------------------------
describe("POST /api/upload", () => {
  beforeEach(() => {
    vi.clearAllMocks();
    // Default: storage succeeds
    vi.mocked(saveFile).mockResolvedValue({
      url: "https://cdn.example.com/uploads/abc123.png",
    });
  });

  // ---------------- Happy path ----------------
  describe("valid uploads", () => {
    it("accepts a valid PNG and returns 201 with url/name/size", async () => {
      const res = await POST(makeUploadRequest([validPng(2048)]));
      const body = await res.json();

      expect(res.status).toBe(201);
      expect(body.url).toMatch(/^https?:\/\//);
      expect(body.name).toBe("photo.png");
      expect(body.size).toBe(2048);
      expect(saveFile).toHaveBeenCalledTimes(1);
    });

    it("accepts a valid JPEG", async () => {
      const jpeg = makeFile("pic.jpg", "image/jpeg", makeBytes(512, JPEG_HEADER));
      const res = await POST(makeUploadRequest([jpeg]));
      expect(res.status).toBe(201);
    });

    it("accepts a valid PDF", async () => {
      const res = await POST(makeUploadRequest([validPdf()]));
      expect(res.status).toBe(201);
    });

    it("passes the file content and MIME type to storage unchanged", async () => {
      await POST(makeUploadRequest([validPng(100)]));

      const [, buffer, mime] = vi.mocked(saveFile).mock.calls[0];
      expect(Buffer.isBuffer(buffer) || buffer instanceof Uint8Array).toBe(true);
      expect((buffer as Uint8Array).length).toBe(100);
      expect(mime).toBe("image/png");
    });

    it("accepts filenames with unicode and spaces", async () => {
      const f = makeFile("รายงาน ปี 2026.pdf", "application/pdf", makeBytes(256, PDF_HEADER));
      const res = await POST(makeUploadRequest([f]));
      expect(res.status).toBe(201);
    });
  });

  // ---------------- Missing / malformed input ----------------
  describe("invalid requests", () => {
    it("returns 400 when no file field is present", async () => {
      const req = new Request("http://localhost/api/upload", {
        method: "POST",
        body: new FormData(), // empty form
      });
      const res = await POST(req);
      expect(res.status).toBe(400);
      expect(saveFile).not.toHaveBeenCalled();
    });

    it("returns 400 when the file is in the wrong field name", async () => {
      const res = await POST(makeUploadRequest([validPng()], "attachment"));
      expect(res.status).toBe(400);
      expect(saveFile).not.toHaveBeenCalled();
    });

    it("returns 400 when the field is a plain string, not a file", async () => {
      const form = new FormData();
      form.append("file", "not-a-file");
      const res = await POST(
        new Request("http://localhost/api/upload", { method: "POST", body: form })
      );
      expect(res.status).toBe(400);
    });

    it("returns 400 (or 415) when body is not multipart", async () => {
      const req = new Request("http://localhost/api/upload", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ file: "x" }),
      });
      const res = await POST(req);
      expect([400, 415]).toContain(res.status);
      expect(saveFile).not.toHaveBeenCalled();
    });

    it("returns 400 for an empty (0 byte) file", async () => {
      const empty = makeFile("empty.png", "image/png", new Uint8Array(0));
      const res = await POST(makeUploadRequest([empty]));
      expect(res.status).toBe(400);
      expect(saveFile).not.toHaveBeenCalled();
    });
  });

  // ---------------- Size boundaries (BVA) ----------------
  describe("size limits (boundary value analysis)", () => {
    it("accepts a 1 byte file (lower boundary + 1)", async () => {
      const res = await POST(makeUploadRequest([validPng(1)]));
      expect(res.status).toBe(201);
    });

    it("accepts MAX_SIZE - 1 bytes", async () => {
      const res = await POST(makeUploadRequest([validPng(MAX_SIZE - 1)]));
      expect(res.status).toBe(201);
    });

    it("accepts exactly MAX_SIZE bytes", async () => {
      const res = await POST(makeUploadRequest([validPng(MAX_SIZE)]));
      expect(res.status).toBe(201);
    });

    it("rejects MAX_SIZE + 1 bytes with 413", async () => {
      const res = await POST(makeUploadRequest([validPng(MAX_SIZE + 1)]));
      expect(res.status).toBe(413);
      expect(saveFile).not.toHaveBeenCalled();
    });
  });

  // ---------------- File type validation ----------------
  describe("file type validation", () => {
    it.each([
      ["script.exe", "application/x-msdownload"],
      ["page.html", "text/html"],
      ["run.sh", "application/x-sh"],
      ["data.svg", "image/svg+xml"], // SVG can carry scripts: common XSS vector
      ["archive.zip", "application/zip"],
    ])("rejects %s (%s) with 415", async (name, type) => {
      const f = makeFile(name, type, makeBytes(256, [0x00]));
      const res = await POST(makeUploadRequest([f]));
      expect(res.status).toBe(415);
      expect(saveFile).not.toHaveBeenCalled();
    });

    it("rejects an .exe renamed to .png when content does not match", async () => {
      // Declared MIME + extension say PNG, but bytes are a Windows executable ("MZ")
      const fake = makeFile("evil.png", "image/png", makeBytes(256, [0x4d, 0x5a]));
      const res = await POST(makeUploadRequest([fake]));
      expect([400, 415]).toContain(res.status);
      expect(saveFile).not.toHaveBeenCalled();
    });

    it("rejects a MIME / extension mismatch (png bytes labelled application/pdf)", async () => {
      const f = makeFile("photo.png", "application/pdf", makeBytes(256, PNG_HEADER));
      const res = await POST(makeUploadRequest([f]));
      expect([400, 415]).toContain(res.status);
    });

    it("rejects a double extension like shell.php.png when content is not an image", async () => {
      const f = makeFile("shell.php.png", "image/png", makeBytes(256, [0x3c, 0x3f])); // "<?"
      const res = await POST(makeUploadRequest([f]));
      expect([400, 415]).toContain(res.status);
    });

    it("treats the extension case-insensitively (PHOTO.PNG is valid)", async () => {
      const f = makeFile("PHOTO.PNG", "image/png", makeBytes(256, PNG_HEADER));
      const res = await POST(makeUploadRequest([f]));
      expect(res.status).toBe(201);
    });
  });

  // ---------------- Filename security ----------------
  describe("filename sanitization", () => {
    it.each([
      "../../etc/passwd.png",
      "..\\..\\windows\\system32\\x.png",
      "/absolute/path/x.png",
      "a/b/c.png",
    ])("never lets path traversal reach storage: %s", async (evilName) => {
      const f = makeFile(evilName, "image/png", makeBytes(256, PNG_HEADER));
      const res = await POST(makeUploadRequest([f]));

      if (res.status === 201) {
        // If accepted, the name handed to storage must be a bare filename
        const [savedName] = vi.mocked(saveFile).mock.calls[0];
        expect(savedName).not.toMatch(/(\.\.|\/|\\)/);
      } else {
        expect(res.status).toBe(400);
        expect(saveFile).not.toHaveBeenCalled();
      }
    });

    it("strips or rejects null bytes in the filename", async () => {
      const f = makeFile("photo.png\u0000.exe", "image/png", makeBytes(256, PNG_HEADER));
      const res = await POST(makeUploadRequest([f]));

      if (res.status === 201) {
        const [savedName] = vi.mocked(saveFile).mock.calls[0];
        expect(savedName).not.toContain("\u0000");
      } else {
        expect([400, 415]).toContain(res.status);
      }
    });

    it("handles a very long filename (300 chars) without crashing", async () => {
      const f = makeFile("a".repeat(300) + ".png", "image/png", makeBytes(256, PNG_HEADER));
      const res = await POST(makeUploadRequest([f]));
      expect([201, 400]).toContain(res.status); // either truncated or rejected, never 500
    });

    it("does not overwrite: two uploads with the same name get different stored names/urls", async () => {
      vi.mocked(saveFile)
        .mockResolvedValueOnce({ url: "https://cdn.example.com/uploads/one.png" })
        .mockResolvedValueOnce({ url: "https://cdn.example.com/uploads/two.png" });

      const a = await (await POST(makeUploadRequest([validPng()]))).json();
      const b = await (await POST(makeUploadRequest([validPng()]))).json();

      expect(a.url).not.toBe(b.url);
    });
  });

  // ---------------- Multiple files ----------------
  describe("multiple files", () => {
    it("rejects more than one file if the endpoint is single-file", async () => {
      // Change to expect 201 + 2 saves if your endpoint supports batches
      const res = await POST(makeUploadRequest([validPng(), validPdf()]));
      expect(res.status).toBe(400);
      expect(saveFile).not.toHaveBeenCalled();
    });
  });

  // ---------------- Failure handling ----------------
  describe("server-side failures", () => {
    it("returns 500 with a generic message when storage throws", async () => {
      vi.mocked(saveFile).mockRejectedValue(new Error("S3: AccessDenied arn:aws:..."));

      const res = await POST(makeUploadRequest([validPng()]));
      const text = await res.text();

      expect(res.status).toBe(500);
      // Internal error details must not leak to the client
      expect(text).not.toMatch(/AccessDenied|arn:aws/i);
    });

    it("does not report success if storage returns no url", async () => {
      vi.mocked(saveFile).mockResolvedValue({ url: "" });
      const res = await POST(makeUploadRequest([validPng()]));
      expect(res.status).toBe(500);
    });
  });
});
```
